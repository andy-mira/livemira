cmake_minimum_required(VERSION 3.20)
project(WebARCompute LANGUAGES CXX)
add_library(webar_compute STATIC common/preprocess.cpp)
target_compile_features(webar_compute PUBLIC cxx_std_17)
target_include_directories(webar_compute PUBLIC common)
if(MSVC)
  target_compile_options(webar_compute PUBLIC /EHsc)
endif()
include(CTest)
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  add_executable(compute_reference_test ../tests/compute/native-reference.cpp)
  target_link_libraries(compute_reference_test PRIVATE webar_compute)
  add_test(NAME compute_reference COMMAND compute_reference_test)
endif()
